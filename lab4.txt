#!/bin/bash
if [ $# -le 0 ]; then
echo "Arguments are not entered"
else
for file in $*
do
echo "Original Name: $file"
newname=$(echo -n "$file" |tr "[a-z]" "[A-Z]")
mv "$file" "$newname"
echo "New names: $newname"
done
fi

file="$1"

set -- $(ls -i "$file")
