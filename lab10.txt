#!/usr/bin/awk -f
{
split($0, arr, "-")
if ((arr[2] < 1) || (arr[2] > 12) || (arr[1] < 1) || (arr[1] > 31))
{
print "invalid date"
exit 1
}
else
{
if (arr[2] == 1) m = "Jan"
if (arr[2] == 2) m = "Feb"
if (arr[2] == 3) m = "March"
if (arr[2] == 4) m = "April"
if (arr[2] == 5) m = "May"
if (arr[2] == 6) m = "Jun"
if (arr[2] == 7) m = "July"
if (arr[2] == 8) m = "Aug"
if (arr[2] == 9) m = "Sept"
if (arr[2] == 10) m = "Oct"
if (arr[2] == 11) m = "Nov"
if (arr[2] == 12) m = "Dec"
print m, arr[1], arr[3]
}
}
