// Circular Queue Operations

#include <stdio.h>
#include <stdlib.h>

#define MAX 5  // Maximum capacity of the circular queue

// Global variables
int queue[MAX];
int front = 0;
int rear = -1;
int count = 0; // Tracks the current number of elements

// Function Declarations
void CQ_enqueue();
void CQ_dequeue();
void CQ_display();

int main() {
    int choice;

    while (1) {
        printf("\n--- Circular Queue Menu ---\n");
        printf("1. CQ_Enqueue (Insert Element)\n");
        printf("2. CQ_Dequeue (Delete Element)\n");
        printf("3. CQ_Display Queue\n");
        printf("4. Exit\n");
        printf("Enter your choice (1-4): ");
        
        // Reading choice from keyboard
        if (scanf("%d", &choice) != 1) {
            printf("Invalid input. Exiting.\n");
            break;
        }

        switch (choice) {
            case 1:
                CQ_enqueue();
                break;
            case 2:
                CQ_dequeue();
                break;
            case 3:
                CQ_display();
                break;
            case 4:
                printf("Exiting program.\n");
                exit(0);
            default:
                printf("Invalid choice! Please select between 1 and 4.\n");
        }
    }
    return 0;
}

// Inserts an element into the circular queue
void CQ_enqueue() {
    int value;

    // Check overflow using count
    if (count == MAX) {
        printf("Queue Overflow! The queue is completely full.\n");
        return;
    }

    printf("Enter the element to insert: ");
    scanf("%d", &value);

    // Circular increment of rear pointer
    rear = (rear + 1) % MAX;
    queue[rear] = value;
    count++; // Increment the element counter
    printf("Successfully inserted %d into the queue.\n", value);
}

// Removes an element from the circular queue
void CQ_dequeue() {
    // Check underflow using count
    if (count == 0) {
        printf("Queue Underflow! The queue is empty.\n");
        return;
    }

    // Retrieve and display front element
    printf("Dequeued element: %d\n", queue[front]);

    // Circular increment of front pointer
    front = (front + 1) % MAX;
    count--; // Decrement the element counter
}

// Displays the active elements inside the queue
void CQ_display() {
    if (count == 0) {
        printf("Queue is empty nothing to show.\n");
        return;
    }

    printf("Current Queue status (Total Elements: %d):\n", count);
    
    int index = front;
    for (int i = 0; i < count; i++) {
        printf("%d ", queue[index]);
        index = (index + 1) % MAX; // Circular wrap-around
    }
    printf("\n");
}
