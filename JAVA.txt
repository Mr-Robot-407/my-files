import java.util.LinkedList; 
import java.util.Queue; 
  
class SharedBuffer { 
    private final Queue<Integer> buffer = new LinkedList<>(); 
    private final int CAPACITY = 5; 
  
    public synchronized void produce(int value) throws InterruptedException { 
        while (buffer.size() == CAPACITY) { 
            System.out.println("Buffer full. Producer waiting..."); 
            wait(); 
        } 
        buffer.add(value); 
        System.out.println("Produced: " + value); 
        notify();  // wake up a waiting consumer 
    } 
  
    public synchronized int consume() throws InterruptedException { 
        while (buffer.isEmpty()) { 
            System.out.println("Buffer empty. Consumer waiting..."); 
            wait(); 
        } 
        int value = buffer.poll(); 
        System.out.println("Consumed: " + value); 
        notify();  // wake up a waiting producer 
        return value; 
    } 
} 
  
class Producer implements Runnable { 
    private SharedBuffer sharedBuffer; 
  
    Producer(SharedBuffer sharedBuffer) { 
        this.sharedBuffer = sharedBuffer; 
    } 
  
    public void run() { 
        try { 
            for (int i = 1; i <= 8; i++) { 
                sharedBuffer.produce(i); 
                Thread.sleep(30); 
            } 
        } catch (InterruptedException e) { 
            Thread.currentThread().interrupt(); 
        } 
    } 
}

    private SharedBuffer sharedBuffer;

        this.sharedBuffer = sharedBuffer;

public class Lab6 { 
    public static void main(String[] args) throws InterruptedException { 
        SharedBuffer sharedBuffer = new SharedBuffer(); 
  
        Thread producerThread = new Thread(new Producer(sharedBuffer), "Producer"); 
        Thread consumerThread = new Thread(new Consumer(sharedBuffer), "Consumer"); 
  
        producerThread.start(); 
        consumerThread.start(); 
  
        producerThread.join(); 
        consumerThread.join();

import javafx.application.Application; 
import javafx.scene.Scene; 
import javafx.scene.control.*; 
import javafx.scene.layout.VBox; 
import javafx.stage.Stage; 
public class ControlsDemo extends Application { 
    @Override 
    public void start(Stage stage) { 
        Label nameLabel = new Label("Enter your name:"); 
        TextField nameField = new TextField(); 
        Label outputLabel = new Label(); 
        Button button = new Button("Submit"); 
        button.setOnAction(e -> { 
            String name = nameField.getText(); 
            if (name.isEmpty()) { 
                Alert alert = new Alert(Alert.AlertType.WARNING); 
                alert.setTitle("Warning"); 
                alert.setHeaderText(null); 
                alert.setContentText("Please enter your name!"); 
                alert.showAndWait(); 
            } else { 
                outputLabel.setText("Welcome, " + name + "!"); 
                Alert alert = new Alert(Alert.AlertType.INFORMATION); 
                alert.setTitle("Message"); 
                alert.setHeaderText("Success"); 
                alert.setContentText("Your name was submitted successfully."); 
                alert.showAndWait(); 
            } 
        }); 
        VBox root = new VBox(10); 
        root.getChildren().addAll( 
            nameLabel, nameField, button, outputLabel 
        ); 
        Scene scene = new Scene(root, 350, 250); 
        stage.setTitle("Student Greeting Application"); 
        stage.setScene(scene); 
        stage.show(); 
    } 
    public static void main(String[] args) { 
        launch(args); 
    } 
}

import java.io.*; 
  
class Student implements Serializable { 
    int rollNo; 
    String name; 
    double marks; 
  
    Student(int rollNo, String name, double marks) { 
        this.rollNo = rollNo; 
        this.name = name; 
        this.marks = marks; 
    } 
  
    @Override 
    public String toString() { 
        return rollNo + "," + name + "," + marks; 
    } 
} 
  
public class Lab10 { 
    public static void main(String[] args) { 
        String fileName = "students.txt"; 
  
        // ---- Sequential Write ---- 
        try (BufferedWriter bw = new BufferedWriter(new FileWriter(fileName))) { 
            Student[] students = { 
                new Student(1, "Nethravathi", 88.5), 
                new Student(2, "Kusumitha", 91.0), 
                new Student(3, "Govind", 76.25) 
            }; 
            for (Student s : students) { 
                bw.write(s.toString()); 
                bw.newLine(); 
            } 
            System.out.println("Records written sequentially to " + fileName); 
        } catch (IOException e) { 
            System.out.println("Error writing to file: " + e.getMessage()); 
        } 
  
        // ---- Sequential Read (record by record, in order) ---- 
        System.out.println("\nReading records sequentially:"); 
        try (BufferedReader br = new BufferedReader(new FileReader(fileName))) { 
            String line; 
            int recordNum = 1; 
            while ((line = br.readLine()) != null) { 
                String[] fields = line.split(","); 
                System.out.println("Record " + recordNum + " -> Roll No: " + fields[0] 
                        + ", Name: " + fields[1] + ", Marks: " + fields[2]); 
                recordNum++; 
            } 
        } catch (IOException e) { 
            System.out.println("Error reading file: " + e.getMessage()); 
        } 
  
        // ---- Append a new record (sequential access continues at end) ---- 
        try (BufferedWriter bw = new BufferedWriter(new FileWriter(fileName, true))) {
