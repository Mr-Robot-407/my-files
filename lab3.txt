#!/bin/bash
trap '' 2 20 3
stty -echo eof undef

echo -n "Enter password to lock terminal: "
read -r pass1
echo
echo -n "Re-enter password to confirm:"
read -r pass2
echo

if [ "$pass1" != "$pass2" ]; then
echo "Passwords do not match! Terminal not locked."
stty echo eof '^D'
exit 1
fi

echo "Terminal is locked."
while true; do
echo -n "Enter password to unlock: "
read -r pass_try
echo
if [ "$pass_try" = "$pass1" ]; then
echo "Terminal unlocked successfully."
break
else
echo "Incorrect password. Try again."
fi
done
stty echo eof '^D'
