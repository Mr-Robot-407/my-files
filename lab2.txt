#!/bin/bash
if [ $# -lt 2 ]; then
echo "Error: Atleast two args should be there"
else
x=`ls -l $1|cut -c 2-10`
echo $x
y=`ls -l $2|cut -c 2-10`
echo $y
if [ $x == $y ]; then
echo "Permissions are same"
echo $x
else
echo "Permissions are not same"
echo $x
echo $y
fi
fi
