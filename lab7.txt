#!/bin/bash
hour=$(date +%k)
echo "Hour: $hour"
if [ $hour -lt 12 ]; then
echo "Good morning"
elif [ $hour -lt 17 ]; then
echo "Good Afternoon"
else
echo "Good Evening"
fi
