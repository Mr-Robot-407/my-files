// Circular Linked List Program

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

struct node *last = NULL;

// Function declarations
void beg_insert();
void last_insert();
void random_insert();
void begin_delete();
void last_delete();
void random_delete();
void search();
void display();

int main() {
    int choice;

    while (1) {
        printf("\n\n******** CIRCULAR LINKED LIST ********\n");
        printf("1. Insert at Beginning\n");
        printf("2. Insert at End\n");
        printf("3. Insert at Any Position\n");
        printf("4. Delete from Beginning\n");
        printf("5. Delete from End\n");
        printf("6. Delete from Any Position\n");
        printf("7. Search Element\n");
        printf("8. Display List\n");
        printf("9. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1: beg_insert(); break;
            case 2: last_insert(); break;
            case 3: random_insert(); break;
            case 4: begin_delete(); break;
            case 5: last_delete(); break;
            case 6: random_delete(); break;
            case 7: search(); break;
            case 8: display(); break;
            case 9: exit(0);
            default: printf("\nInvalid Choice! Try again.\n");
        }
    }
    return 0;
}

// Insert a node at the beginning of the circular list
void beg_insert() {
    struct node *ptr, *temp;
    int item;

    ptr = (struct node*)malloc(sizeof(struct node));
    if (ptr == NULL) {
        printf("Memory Overflow!\n");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &item);
    ptr->data = item;

    if (last == NULL) {
        last = ptr;
        ptr->next = last;
    } else {
        temp = last;
        while (temp->next != last) {
            temp = temp->next;
        }
        ptr->next = last;
        temp->next = ptr;
        last = ptr; // New node becomes the first node
    }
    printf("Node Inserted at Beginning\n");
}

// Insert a node at the end of the circular list
void last_insert() {
    struct node *ptr, *temp;
    int item;

    ptr = (struct node*)malloc(sizeof(struct node));
    if (ptr == NULL) {
        printf("Memory Overflow!\n");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &item);
    ptr->data = item;

    if (last == NULL) {
        last = ptr;
        ptr->next = last;
    } else {
        temp = last;
        while (temp->next != last) {
            temp = temp->next;
        }
        temp->next = ptr;
        ptr->next = last;
    }
    printf("Node Inserted at End\n");
}

// Insert a node after a given position
void random_insert() {
    struct node *ptr, *temp;
    int item, loc, i;

    ptr = (struct node*)malloc(sizeof(struct node));
    if (ptr == NULL) {
        printf("Memory Overflow!\n");
        return;
    }

    printf("Enter value: ");
    scanf("%d", &item);
    ptr->data = item;

    printf("Enter position after which to insert: ");
    scanf("%d", &loc);

    temp = last;
    for (i = 1; i < loc; i++) {
        temp = temp->next;
        if (temp == last) {
            printf("Invalid Position!\n");
            free(ptr);
            return;
        }
    }
    ptr->next = temp->next;
    temp->next = ptr;
    printf("Node Inserted\n");
}

// Delete the first node
void begin_delete() {
    struct node *temp, *tail;

    if (last == NULL) {
        printf("List is empty\n");
        return;
    }

    // Only one node in the list
    if (last->next == last) {
        free(last);
        last = NULL;
    } else {
        tail = last;
        while (tail->next != last) {
            tail = tail->next;
        }
        temp = last;
        last = last->next;
        tail->next = last;
        free(temp);
    }
    printf("Node Deleted from Beginning\n");
}

// Delete the last node
void last_delete() {
    struct node *temp, *prev;

    if (last == NULL) {
        printf("List is empty\n");
        return;
    }

    // Only one node in the list
    if (last->next == last) {
        free(last);
        last = NULL;
    } else {
        temp = last;
        while (temp->next != last) {
            prev = temp;
            temp = temp->next;
        }
        prev->next = last;
        free(temp);
    }
    printf("Node Deleted from End\n");
}

// Delete a node from a specific position
void random_delete() {
    struct node *temp, *prev;
    int loc, i;

    if (last == NULL) {
        printf("List is empty\n");
        return;
    }

    printf("Enter position to delete: ");
    scanf("%d", &loc);

    if (loc == 1) {
        begin_delete();
        return;
    }

    temp = last;
    for (i = 1; i < loc; i++) {
        prev = temp;
        temp = temp->next;
        if (temp == last) {
            printf("Invalid Position!\n");
            return;
        }
    }
    prev->next = temp->next;
    free(temp);
    printf("Node Deleted at position %d\n", loc);
}

// Search for an element
void search() {
    struct node *temp;
    int item, pos = 1, found = 0;

    if (last == NULL) {
        printf("List is empty\n");
        return;
    }

    printf("Enter element to search: ");
    scanf("%d", &item);

    temp = last;
    do {
        if (temp->data == item) {
            printf("Element found at position %d\n", pos);
            found = 1;
        }
        temp = temp->next;
        pos++;
    } while (temp != last);

    if (!found) {
        printf("Element not found in list\n");
    }
}

// Display elements in the circular list
void display() {
    struct node *temp;

    if (last == NULL) {
        printf("List is empty\n");
        return;
    }

    temp = last;
    printf("\nCircular Linked List: ");
    do {
        printf("%d -> ", temp->data);
        temp = temp->next;
    } while (temp != last);
    printf("(head)\n");
}
